cmake_minimum_required(VERSION 3.8)
project(eight_trajectory)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(eight_trajectory src/eight_trajectory.cpp)
target_compile_features(eight_trajectory PUBLIC cxx_std_17)
ament_target_dependencies(eight_trajectory rclcpp nav_msgs std_msgs)
install(TARGETS eight_trajectory DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
